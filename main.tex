% One bullet point.
\newcommand{\resumeItem}[1]{
    \item\small{
        {#1 \vspace{-2pt}}
    }
}

% A two-line heading: bold line on top, italic line underneath.
%   #1 = left bold    (role / school / project title)
%   #2 = right bold   (dates)
%   #3 = left italic  (company / course)
%   #4 = right italic (location)
\newcommand{\resumeSubheading}[4]{
    \vspace{-2pt}\item
        \begin{tabular*}{1.0\textwidth}[t]{l@{\extracolsep{\fill}}r}
            \textbf{#1} & \textbf{\small #2} \\
            \textit{\small#3} & \textit{\small #4} \\
        \end{tabular*}\vspace{-7pt}
}

% A single-line heading (used for projects): bold title left, date right.
\newcommand{\resumeProjectHeading}[2]{
        \item
        \begin{tabular*}{1.001\textwidth}{l@{\extracolsep{\fill}}r}
            \small#1 & \textbf{\small #2}\\
        \end{tabular*}\vspace{-7pt}
}
